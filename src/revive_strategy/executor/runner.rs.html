47    /// Sets the balance of an account.
48    ///
49    /// Amount should be in the range of [0, u128::MAX] despite the type
50    /// because Ethereum balances are u256 while Polkadot balances are u128.
51    fn set_balance(
52        &self,
53        executor: &mut foundry_evm::executors::Executor,
54        address: Address,
55        amount: U256,
56    ) -> foundry_evm::backend::BackendResult<()> {
57        let amount_pvm =
58            sp_core::U256::from_little_endian(&amount.as_le_bytes()).min(u128::MAX.into());
59        let amount_pvm = U256ToBalance::convert(amount_pvm);
60        let amount_evm = U256::from(amount_pvm);
61        if amount != amount_evm {
62            let _ = sh_err!("Amount mismatch {amount} != {amount_evm}, Polkadot balances are u128. Test results may be incorrect.");
63        }
64        EvmExecutorStrategyRunner.set_balance(executor, address, amount_evm)?;
65
66        let backend = get_backend_ref(executor.backend().strategy.context.as_ref());
67        let mut ext = backend.revive_test_externalities.lock().unwrap();
68        ext.execute_with(|| {
69            pallet_balances::Pallet::<Runtime>::set_balance(
70                &AccountId::to_fallback_account_id(&H160::from_slice(address.as_slice())),
71                amount_pvm,
72            );
73        });
74        Ok(())
75    }
76
77    fn get_balance(
78        &self,
79        executor: &foundry_evm::executors::Executor,
80        address: Address,
81    ) -> foundry_evm::backend::BackendResult<U256> {
82        let evm_balance = EvmExecutorStrategyRunner.get_balance(executor, address)?;
83
84        let backend = get_backend_ref(executor.backend().strategy.context.as_ref());
85        let mut ext = backend.revive_test_externalities.lock().unwrap();
86        let balance = ext.execute_with(|| {
87            pallet_balances::Pallet::<Runtime>::free_balance(AccountId::to_fallback_account_id(
88                &H160::from_slice(address.as_slice()),
89            ))
90        });
91        assert_eq!(evm_balance, U256::from(balance));
92        Ok(evm_balance)
93    }

95    fn set_nonce(
96        &self,
97        executor: &mut foundry_evm::executors::Executor,
98        address: Address,
99        nonce: u64,
100    ) -> foundry_evm::backend::BackendResult<()> {
101        EvmExecutorStrategyRunner.set_nonce(executor, address, nonce)?;
102        let backend = get_backend_ref(executor.backend().strategy.context.as_ref());
103        let mut ext = backend.revive_test_externalities.lock().unwrap();
104        ext.execute_with(|| {
105            let account_id =
106                AccountId::to_fallback_account_id(&H160::from_slice(address.as_slice()));
107            let current_nonce = System::account_nonce(&account_id);
108
109            assert!(
110                current_nonce as u64 <= nonce,
111                "Cannot set nonce lower than current nonce: {current_nonce} > {nonce}"
112            );
113
114            while (System::account_nonce(&account_id) as u64) < nonce {
115                System::inc_account_nonce(&account_id);
116            }
117        });
118        Ok(())
119    }
120
121    fn get_nonce(
122        &self,
123        executor: &foundry_evm::executors::Executor,
124        address: Address,
125    ) -> foundry_evm::backend::BackendResult<u64> {
126        let evm_nonce = EvmExecutorStrategyRunner.get_nonce(executor, address)?;
127        let backend = get_backend_ref(executor.backend().strategy.context.as_ref());
128        let mut ext = backend.revive_test_externalities.lock().unwrap();
129        let revive_nonce = ext.execute_with(|| {
130            System::account_nonce(AccountId::to_fallback_account_id(&H160::from_slice(
131                address.as_slice(),
132            )))
133        });
134
135        assert_eq!(evm_nonce, revive_nonce as u64);
136        Ok(evm_nonce)
137    }

139    fn call(
140        &self,
141        ctx: &dyn ExecutorStrategyContext,
142        backend: &mut foundry_evm::backend::CowBackend<'_>,
143        env: &mut EnvWithHandlerCfg,
144        executor_env: &EnvWithHandlerCfg,
145        inspector: &mut foundry_evm::inspectors::InspectorStack,
146    ) -> eyre::Result<ResultAndState> {
147        let ctx = get_context_ref(ctx);
148        if ctx.wip_in_pvm {
149            backend.inspect(env, inspector, Box::new(ReviveInspectContext))
150        } else {
151            EvmExecutorStrategyRunner.call(ctx, backend, env, executor_env, inspector)
152        }
153    }
154
155    fn transact(
156        &self,
157        ctx: &mut dyn ExecutorStrategyContext,

159        env: &mut EnvWithHandlerCfg,

161        inspector: &mut foundry_evm::inspectors::InspectorStack,
162    ) -> eyre::Result<ResultAndState> {
163        let ctx = get_context_ref_mut(ctx);
164        if ctx.wip_in_pvm {
165            backend.inspect(env, inspector, Box::new(ReviveInspectContext))
166        } else {
167            EvmExecutorStrategyRunner.transact(ctx, backend, env, executor_env, inspector)
168        }
169    }
