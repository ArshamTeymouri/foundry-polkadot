1//! This crate provides the Revive strategy for the Foundry EVM ExecutorStrategy.
2//!
3//! It is designed to work with the Revive runtime, allowing for the execution of smart contracts
4//! in a Polkadot environment.
5//!
6//! It is heavily inspired from <https://github.com/matter-labs/foundry-zksync/tree/main/crates/strategy/zksync>
7use foundry_evm::executors::ExecutorStrategy;
8
9use crate::executor::{
10    context::ReviveExecutorStrategyContext, runner::ReviveExecutorStrategyRunner,
11};
12
13mod backend;
14mod executor;
15
16/// Create Revive strategy for [ExecutorStrategy].
17pub trait ReviveExecutorStrategyBuilder {
18    /// Create new revive strategy.
19    fn new_revive() -> Self;
20}
21
22impl ReviveExecutorStrategyBuilder for ExecutorStrategy {
23    fn new_revive() -> Self {
24        Self {
25            runner: &ReviveExecutorStrategyRunner,
26            context: Box::new(ReviveExecutorStrategyContext::default()),
27        }
28    }
29}
